💡 Recomendaciones para tu App

Para Gastos App, te recomiendo:

  • Opción 1 (Minimalista): Simple Spinner + texto "Guardando..."
  • Opción 2 (iOS Style - Recomendado): Dots Bounce + texto
  • Opción 3 (Moderno): Material Spinner + texto

¿Cuándo usar cada uno?

  • Acciones rápidas (< 2 seg): Simple Spinner
  • Acciones normales (2-5 seg): Dots Bounce
  • Acciones largas (> 5 seg): Material Spinner con texto descriptivo

📝 Código de Ejemplo (React)

// Componente LoadingSpinner.tsx
interface LoadingSpinnerProps {
  variant?: 'simple' | 'dots' | 'material';
  size?: 'sm' | 'md' | 'lg';
}

export function LoadingSpinner({ variant = 'dots', size = 'md' }: LoadingSpinnerProps) {
  const sizes = {
    sm: 'w-3 h-3',
    md: 'w-4 h-4',
    lg: 'w-5 h-5',
  };

  if (variant === 'dots') {
    return (
      <div className="flex gap-1">
        <span className="w-1.5 h-1.5 bg-current rounded-full animate-bounce [animation-delay:-0.32s]"></span>
        <span className="w-1.5 h-1.5 bg-current rounded-full animate-bounce [animation-delay:-0.16s]"></span>
        <span className="w-1.5 h-1.5 bg-current rounded-full animate-bounce"></span>
      </div>
    );
  }

  // Simple spinner
  return (
    <div className={`${sizes[size]} border-2 border-current border-t-transparent rounded-full animate-spin`}></div>
  );
}

// Uso en botones
<button disabled={loading} className="btn btn-primary">
  {loading ? (
    <>
      <LoadingSpinner variant="dots" />
      <span>Guardando...</span>
    </>
  ) : (
    'Guardar'
  )}
</button>
